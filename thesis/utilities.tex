% blank page helper used by main.tex
\newcommand\myemptypage{%
  \begingroup
  \hypersetup{pageanchor=false}%
  \null
  \thispagestyle{empty}%
  \addtocounter{page}{-1}%
  \newpage
  \hypersetup{pageanchor=true}%
  \endgroup
}
